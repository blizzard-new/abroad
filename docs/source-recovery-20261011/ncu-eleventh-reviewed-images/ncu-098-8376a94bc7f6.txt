function konumlariTabloyaEkle(konumListesi) {
    var html = '';

    konumListesi.forEach(konum => {
        html += '
';
        

        if (konum["birim-en"] != "") {
            html += '
' + konum.birim + '
' + konum["birim-en"] + '
';
        } else {
            html += '
' + konum.birim + '
';
        }

        html += '
' + konum.telefon + '
';
        html += '
';
        html += '
Konuma Git
';
        html += '
';
    });

    document.getElementById("results-list").innerHTML = html;
}
